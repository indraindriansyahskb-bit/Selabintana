function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); try { var data = JSON.parse(e.postData.contents); var action = data.action; if (action === "login") { var sheetUsers = ss.getSheetByName("Users"); var valuesUsers = sheetUsers.getDataRange().getValues(); for (var i = 1; i < valuesUsers.length; i++) { if (valuesUsers[i][0] == data.username && valuesUsers[i][1] == data.password) { return responseJSON({ status: "success", message: "Login Berhasil", user: { nama: valuesUsers[i][2], divisi: valuesUsers[i][3], role: valuesUsers[i][4] } }); } } return responseJSON({ status: "error", message: "Username atau Password salah!" }); } else if (action === "submit_permintaan_manual") { var sheetReq = ss.getSheetByName("Permintaan_Barang"); var kodeBarang = "BRG-" + Math.floor(1000 + Math.random() * 9000); // Generate kode unik otomatis var tanggal = new Date().toLocaleString("id-ID", { timeZone: "Asia/Jakarta" }); sheetReq.appendRow([ kodeBarang, tanggal, data.nama_pemohon, data.divisi, data.nama_barang, data.jumlah, data.satuan, data.supplier, data.keterangan, "Pending" ]); return responseJSON({ status: "success", message: "Permintaan barang berhasil diajukan!", kode: kodeBarang }); } } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e.parameter.action; if (action === "get_permintaan") { var sheetReq = ss.getSheetByName("Permintaan_Barang"); var valuesReq = sheetReq.getDataRange().getValues(); var resultReq = []; for (var j = 1; j < valuesReq.length; j++) { resultReq.push({ kode: valuesReq[j][0], tanggal: valuesReq[j][1], pemohon: valuesReq[j][2], divisi: valuesReq[j][3], barang: valuesReq[j][4], jumlah: valuesReq[j][5], satuan: valuesReq[j][6], supplier: valuesReq[j][7], keterangan: valuesReq[j][8], status: valuesReq[j][9] }); } return responseJSON({ status: "success", data: resultReq.reverse() }); } return responseJSON({ status: "error", message: "Aksi tidak valid" }); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
