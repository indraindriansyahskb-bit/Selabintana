Panduan Integration Backend Google Apps Script (GAS)

Aplikasi ini dirancang untuk terhubung langsung dengan **Google Sheets** menggunakan Apps Script sebagai API backend. Ikuti langkah di bawah ini untuk deploy:

  1. Buat **Google Sheet Baru** di Google Drive Anda. Buat 4 Tab Sheet dengan nama tepat: Master_Unit, Inventory, Mutasi, dan Users.
  2. Buka menu Ekstensi > Apps Script di Google Sheet tersebut.
  3. Salin (copy) kode Apps Script di bawah ini dan tempelkan ke editor Apps Script.
  4. Klik Deploy > Deployments baru, pilih jenis Web App, atur Who has access menjadi Anyone.
  5. Salin URL Web App yang dihasilkan dan masukkan ke konfigurasi sistem ini.
Kode Apps Script (Code.gs):
// Kode Backend Google Apps Script untuk Sistem Stok Farm
function doGet(e) {
  var action = e.parameter.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (action == 'getInventory') {
    var sheet = ss.getSheetByName('Inventory');
    var data = sheet.getDataRange().getValues();
    return responseJSON(data);
  }
  if (action == 'getMutations') {
    var sheet = ss.getSheetByName('Mutasi');
    var data = sheet.getDataRange().getValues();
    return responseJSON(data);
  }
}

function doPost(e) {
  var contents = JSON.parse(e.postData.contents);
  var action = contents.action;
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // Handle Pengajuan Mutasi Baru
  if (action == 'addMutation') {
    var sheet = ss.getSheetByName('Mutasi');
    sheet.appendRow([
      contents.id, new Date(), contents.itemId, contents.itemName,
      contents.fromUnit, contents.fromGudang, contents.toUnit, contents.toGudang,
      contents.qty, 'PENDING', contents.user
    ]);
    return responseJSON({status: 'SUCCESS', message: 'Mutasi berhasil diajukan'});
  }

  // Handle Approval Superadmin
  if (action == 'approveMutation') {
    var sheetMut = ss.getSheetByName('Mutasi');

    // ...
    return responseJSON({status: 'SUCCESS', message: 'Mutasi disetujui & Stok diperbarui'});
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
